# Посчитать количество всех зарегистрированных акаунтов на сайте, которые мы можем закрепить за кафе, на 30 ноября 2024

SELECT
   point_id,
   COUNT( DISTINCT number ) as count
FROM
   jaco_site_rolls.site_orders_new son
WHERE
   son.`date`<="2024-11-30"
      AND
   `client_id`>0
      AND
   type_order IN (1,2)
GROUP BY
   point_id;

# Количество активных акаунтов – Зарегистрированные аккаунты, которые сделали заказ в ноябре 2024

SELECT
   point_id,
   COUNT(*) as count
FROM(
   SELECT
      son.`number`,
      son.`point_id`
   FROM
      jaco_site_rolls.`site_orders_new` son
      LEFT JOIN jaco_main_rolls.`site_users` su
         ON
            son.`client_id`=su.`id`
      WHERE
         son.`date` BETWEEN "2024-11-01" AND "2024-11-29"
            AND
         son.`client_id`>0
            AND
         son.type_order IN (1,2)
      GROUP BY
         son.`number`
) as t
GROUP BY
   point_id

# количество повторных заказов по промику

SELECT
   t.`number`,
   COUNT(*) as count
FROM (
   SELECT
      son1.`number`
   FROM
      jaco_site_rolls.`site_orders_new` son1
   WHERE
      son1.`promo_name` LIKE 'УДОБНО'
   GROUP BY
      son1.`number`
) as t
LEFT JOIN jaco_site_rolls.`site_orders_new` son2
   ON
      son2.`number`=t.`number`
GROUP BY
   t.`number`
HAVING
   count>3;

# Конверсия онлайн заказов

SELECT

  AVG(100 / ( all_count / is_pay )) as percent_pay

FROM (

  SELECT

  SUM( IF( `status_order`>0, 1, 0 ) ) as is_pay,

     SUM( IF( `status_order`=0, 1, 0 ) ) as is_not_pay,
     COUNT(*) as all_count,
     DATE_FORMAT(`date_time_order`, '%Y-%m') as year_month_

FROM

jaco_rolls_8.`orders`

WHERE

`date_time_order` BETWEEN "2020-01-02 00:00:00" AND "2025-03-09 23:59:59"

GROUP BY

year_month_

) as t;
